# Voice Q&A (Push-to-Talk) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Кнопка микрофона в Race-view — пользователь задаёт короткий вопрос голосом во время гонки, получает короткий голосовой ответ по текущему контексту гонки, без риска перебить critical-озвучку.

**Architecture:** `voice/listener.py` пишет ~5с с микрофона (инъектируемый recorder) → `yandex_ai/stt.py` распознаёт (Yandex SpeechKit STT v1) → `commentator/query.py` строит fact-only промпт и зовёт `AIProvider` с собственным таймаутом → `core/engine.py` оркестрирует в фоновом потоке, пишет статус в `state["voice_query"]`, озвучивает ответ `voice.say(..., "normal")`. Новый `TTSQueue.critical_active`/`Voice.is_critical_active` даёt engine явный гейт «не отвечаем поверх критики» — busy вместо запуска конвейера.

**Tech Stack:** Python 3.12, aiohttp (Yandex Cloud), sounddevice (запись), pytest; фронт — Next/React (`NewSpotterUI`), Bottle API.

**Спека:** [`docs/superpowers/specs/2026-07-01-voice-qa-design.md`](../specs/2026-07-01-voice-qa-design.md) — 3 правки пользователя (push-to-talk базовый; critical-гейт → busy; таймаут-фолбэк + инъектируемый recorder с первого дня) уже внутри этого плана.

```python
"""
commentator/query.py
=====================
Voice Q&A: короткий фактологический ответ на вопрос пользователя по текущему
контексту гонки (тот же timeline+analytics, что видит комментатор).

LLM-вызов ограничен ПО ВРЕМЕНИ отдельно от общего таймаута Yandex-транспорта
(VOICE_ANSWER_TIMEOUT_S) — голосовой диалог требует более быстрой реакции, чем
фоновый ambient-комментарий; если генерация не успела — не ждём, отдаём
FALLBACK_ANSWER (осиротевший поток доиграет в фоне и тихо потеряет результат).
"""
from __future__ import annotations

import logging
import threading

import config

_log = logging.getLogger(__name__)

FALLBACK_ANSWER = "Пока нет уверенного сигнала."


def build_prompt(question: str, context: str, gp_context: str | None = None) -> str:
    parts = [
        "Ты голосовой ассистент гонщика во время заезда. Ответь ОДНОЙ короткой фразой "
        f"(не больше {config.VOICE_ANSWER_MAX_WORDS} слов) СТРОГО по данным контекста "
        "гонки ниже — ничего не выдумывай. Если по вопросу нет уверенных данных, ответь "
        f"ровно: «{FALLBACK_ANSWER}». Без markdown, кавычек и эмодзи.",
        f"\nКОНТЕКСТ ГОНКИ:\n{context}",
    ]
    if gp_context:
        parts.append(f"\nСверка с реальным Гран-при:\n{gp_context}")
    parts.append(f"\nВОПРОС: {question}")
    return "\n".join(parts)


def _cap_words(text: str, max_words: int) -> str:
    words = text.split()
    if len(words) > max_words:
        text = " ".join(words[:max_words])
    return text[:140].rstrip()


def answer_question(question: str, context: str, ai, persona: str,
                    gp_context: str | None = None) -> str:
    """Сгенерировать короткий ответ на вопрос. Всегда что-то возвращает: при
    пустом вопросе, недоступном ИИ, таймауте или пустом ответе — FALLBACK_ANSWER."""
    question = (question or "").strip()
    if not question or ai is None or not getattr(ai, "available", False):
        return FALLBACK_ANSWER

    prompt = build_prompt(question, context, gp_context)
    result: dict[str, str | None] = {"text": None}

    def _run() -> None:
        try:
            result["text"] = ai.generate(prompt, persona)
        except Exception as exc:  # noqa: BLE001
            _log.warning("voice answer generation failed: %s", exc)

    thread = threading.Thread(target=_run, daemon=True, name="voice-answer")
    thread.start()
    thread.join(timeout=config.VOICE_ANSWER_TIMEOUT_S)
    if thread.is_alive():
        _log.warning("voice answer timed out after %.1fs", config.VOICE_ANSWER_TIMEOUT_S)
        return FALLBACK_ANSWER

    text = result["text"]
    if not text or not text.strip():
        return FALLBACK_ANSWER
    return _cap_words(text.strip(), config.VOICE_ANSWER_MAX_WORDS)
```

```python
"""
yandex_ai/stt.py
=================
Yandex SpeechKit STT (v1, короткое аудио): LPCM bytes -> распознанный текст.

Мирроит yandex_ai/speech.py: recognize() — синхронная обёртка, глотает
исключения -> None (вызывающий код деградирует в safe-фолбэк, см. commentator/query.py).
"""
from __future__ import annotations

import json
import logging

import config

_log = logging.getLogger(__name__)


class YandexSTT:
    def __init__(self, client):
        self._client = client

    async def _arecognize(self, audio: bytes, sr: int) -> str | None:
        params = {
            "folderId": self._client.folder_id,
            "lang": "ru-RU",
            "format": "lpcm",
            "sampleRateHertz": str(sr),
        }
        raw = await self._client.post_audio(
            config.YANDEX_STT_URL, audio, params,
            connect=config.YANDEX_STT_CONNECT_TIMEOUT,
            total=config.YANDEX_STT_TOTAL_TIMEOUT,
        )
        if not raw:
            return None
        data = json.loads(raw.decode("utf-8"))
        text = (data.get("result") or "").strip()
        return text or None

    def recognize(self, audio: bytes, sr: int = 48000) -> str | None:
        """Распознать речь из LPCM int16 mono bytes. None — недоступно/сбой/тишина."""
        if not audio:
            return None
        try:
            fut = self._client.submit(self._arecognize(audio, sr))
            return fut.result(timeout=config.YANDEX_STT_TOTAL_TIMEOUT + 1.0)
        except Exception as exc:  # noqa: BLE001
            _log.warning("YandexSTT recognize failed: %s", exc)
            return None
```

```python
"""
voice/listener.py
==================
Push-to-talk запись вопроса с микрофона: sounddevice.rec -> int16 LPCM mono bytes.

recorder инъектируется через конструктор с первого дня (VoiceListener(recorder=...)) —
тесты никогда не трогают реальное аудио-устройство. Нет микрофона / исключение -> None,
вызывающий код (core/engine.py) деградирует в safe-фолбэк без падения приложения.
"""
from __future__ import annotations

import logging

_log = logging.getLogger(__name__)


def _default_recorder(max_sec: float, sr: int) -> bytes | None:
    import sounddevice as sd
    frames = int(max_sec * sr)
    audio = sd.rec(frames, samplerate=sr, channels=1, dtype="int16")
    sd.wait()
    return audio.tobytes()


class VoiceListener:
    def __init__(self, recorder=None):
        self._recorder = recorder or _default_recorder

    def record(self, max_sec: float, sr: int = 48000) -> bytes | None:
        """Записать до max_sec секунд с микрофона. None — нет устройства/сбой."""
        try:
            data = self._recorder(max_sec, sr)
        except Exception as exc:  # noqa: BLE001
            _log.warning("VoiceListener record failed: %s", exc)
            return None
        return data or None
```

```python
    # ------------------------------------------------------------
    # Voice Q&A (push-to-talk)
    # ------------------------------------------------------------

    def ask_voice_question(self) -> dict:
        """Push-to-talk: слушает вопрос, распознаёт, отвечает голосом. Гейты:
        (1) сейчас играет critical-реплика -> busy сразу, конвейер не запускаем
        (не тратим 5-8с записи+STT+LLM на ответ, который лёг бы в очередь позади
        критики); (2) уже идёт другой вопрос -> busy. Иначе — фоновый поток."""
        if self.voice.is_critical_active:
            return {"ok": False, "busy": True, "reason": "critical"}
        with self.state_lock:
            current = self.state.get("voice_query")
            if current is not None and current.get("status") in (
                    "listening", "recognizing", "thinking"):
                return {"ok": False, "busy": True, "reason": "in_progress"}
            self.state["voice_query"] = {"status": "listening", "question": None,
                                         "answer": None, "error": None}
        threading.Thread(target=self._run_voice_question, daemon=True,
                         name="voice-question").start()
        return {"ok": True}

    def _set_voice_query(self, **updates) -> None:
        with self.state_lock:
            vq = dict(self.state.get("voice_query") or {})
            vq.update(updates)
            self.state["voice_query"] = vq

    def _run_voice_question(self) -> None:
        """Слушает -> распознаёт -> отвечает. Фоновый поток (см. ask_voice_question);
        каждый статус пишется в state["voice_query"] для UI-поллинга через /api/state.
        Обёрнуто в try/except: неожиданный сбой должен закрыть busy-guard статусом
        error, а не оставить voice_query подвешенным в thinking/recognizing."""
        try:
            if self._stt is None or not self._yandex_healthy:
                self._set_voice_query(status="error", error="Распознавание недоступно")
                return

            audio = self._voice_listener.record(config.VOICE_QUESTION_MAX_SEC)
            if audio is None:
                self._set_voice_query(status="error", error="Микрофон недоступен")
                return

            self._set_voice_query(status="recognizing")
            question = self._stt.recognize(audio)
            if not question:
                self._set_voice_query(status="error", error="Не расслышал вопрос")
                return

            self._set_voice_query(status="thinking", question=question)
            context = self._build_ai_context({"event_code": "QUESTION"})
            answer = _query.answer_question(question, context, self.ai,
                                            self.commentator.persona,
                                            self.commentator.analytics_context)

            voiced = self._should_voice({"event_code": "USER_Q", "priority": "normal"})
            with self.state_lock:
                self.state["feed"].insert(0, {
                    "time": datetime.now().strftime("%H:%M:%S"),
                    "event_code": "USER_Q",
                    "phrase": f"{question} — {answer}",
                    "color": "#60A5FA", "driver": "",
                    "muted": not voiced, "channel": "commentary"})
                self.state["feed"] = self.state["feed"][:config.MAX_FEED_ITEMS]
            if voiced:
                self.voice.say(answer, priority="normal")
            self._set_voice_query(status="done", answer=answer)
        except Exception as exc:  # noqa: BLE001
            _log.warning("voice question pipeline failed: %s", exc)
            self._set_voice_query(status="error", error="Внутренняя ошибка")
```
